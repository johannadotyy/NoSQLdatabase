//Queries on DB

// patient IDs for medical condition = 'watch'

db.medical_records.find({medical_conditions: 'Watch' }, {patient_id: 1, _id:0})

//patients using band 2 or band 3

db.medical_records.find({ Tracker: {$in: ['Band 2', 'Band 3'] } }, {patient_id: 1, _id: 0 } )

// fitness trackers with rating > 4

db.fitness_trackers.find({ 'Rating (Out of 5)': {$gt: 4} }, {'Brand Name': 1, 'Model Name': 1, _id: 0})



//Optimizing queries on DB with indexes

// patient IDs for medical condition = 'watch'
//optimized with medical condition index

db.medical_records.createIndex({medical_conditions: 1})
db.medical_records.find({medical_conditions: 'Watch' }, {patient_id: 1, _id:0})

//patients using band 2 or band 3
//optimized with index

db.medical_records.createIndex({Tracker: 1})

// fitness trackers with rating > 4
//optimized with index
